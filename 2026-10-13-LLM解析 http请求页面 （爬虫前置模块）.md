用LLM把自然语言解析为固定字段，利用固定字段去拼接url或筛选数据。
例如query字段可以和`baseurl`拼接出完整url（查询参数）`www.amazon.com/s?k=gift+bag`。
利用http库来`get`结果（html界面），再将html界面送到例如crawlee类的爬虫框架中解析。

```python
"""中文要求 -> AI 搜索字段 -> Amazon URL -> 本地 HTML。HTTP 请求使用 requests。"""

import argparse

import getpass

import json

from pathlib import Path

import re

# from urllib.error import HTTPError

from urllib.parse import urlencode

# from urllib.request import Request, urlopen

  

import requests

  
  

PROMPT = '''把用户要求转成一个 Amazon 美国站搜索方向。只返回 JSON，不要解释：

{"query":"英文搜索词","category":"中文商品品类","theme":"中文主题","sort":null}

query 必须保留用户要求的主体和商品品类。

sort 只能是 null（默认相关性）或 "date-desc-rank"（新品优先）。

没有明确要求新品优先时使用 null。不要生成 URL。'''

  
  

def plan_search(requirements, api_key, api_url, model):

    # 第一步：中文要求交给 AI，要求按固定字段填表。

    body = {

        'model': model,

        'messages': [

            {'role': 'system', 'content': PROMPT},

            {'role': 'user', 'content': requirements},

        ],

    }

    headers = {

        'Content-Type': 'application/json',

        'Authorization': 'Bearer ' + api_key,

    }

    # 原来的 urllib 请求方式，保留供对照：

    # request = Request(api_url, data=json.dumps(body).encode('utf-8'), headers=headers)

    # with urlopen(request, timeout=180) as response:

    #     reply = json.load(response)

  

    with requests.post(api_url, json=body, headers=headers, timeout=180) as response:

        response.raise_for_status()

        reply = response.json()

    content = reply['choices'][0]['message']['content'].strip()

    content = re.sub(r'^```(?:json)?\s*|\s*```$', '', content)

    fields = json.loads(content)

    for name in ('query', 'category', 'theme'):

        if not isinstance(fields.get(name), str) or not fields[name].strip():

            raise ValueError(f'模型没有返回有效的 {name}')

    if 'sort' not in fields or fields['sort'] not in (None, 'date-desc-rank'):

        raise ValueError('模型返回的 sort 不符合约定')

    return fields

  
  

def build_url(fields, page):

    # 第二步：Amazon 使用 k 表示关键词、s 表示排序、page 表示页码。

    params = {'k': fields['query'], 'page': page}

    if fields['sort'] is not None:

        params['s'] = fields['sort']

    return 'https://www.amazon.com/s?' + urlencode(params)

  
  

def main():

    parser = argparse.ArgumentParser(description=__doc__)

    parser.add_argument('--requirements', help='中文选品要求；不填时现场输入')

    parser.add_argument('--page', type=int, default=1, choices=range(1, 11))

    parser.add_argument('--api-url', default='https://gptapi.longpean.com/v1/chat/completions')

    parser.add_argument('--model', default='codex-gpt-5.6-terra')

    parser.add_argument('--output', type=Path, default=Path(__file__).parent / 'output')

    args = parser.parse_args()

    requirements = args.requirements or input('中文要求：').strip()

    if not requirements.strip():

        parser.error('中文要求不能为空')

    # 不读取环境变量；交互输入避免把密钥留在命令历史里。

    api_key = getpass.getpass('Longpean client key（输入不显示）：').strip()

    if not api_key:

        parser.error('API Key 不能为空')

  

    print('\n1. 正在让 AI 生成搜索字段……')

    fields = plan_search(requirements, api_key, args.api_url, args.model)

    print(json.dumps(fields, ensure_ascii=False, indent=2))

  

    url = build_url(fields, args.page)

    print('\n2. 拼好的 Amazon URL：\n' + url)

    args.output.mkdir(parents=True, exist_ok=True)

    (args.output / 'plan.json').write_text(json.dumps(fields, ensure_ascii=False, indent=2), encoding='utf-8')

    (args.output / 'url.txt').write_text(url, encoding='utf-8')

  

    # 第三步：请求一次搜索页，保存返回的网页源码。没有自动重试或翻页。

    print('\n3. 正在下载 Amazon HTML……')

    headers = {'User-Agent': 'PatternLoop-InspirationCollector/1.0'}

    # 原来的 urllib 请求方式，保留供对照：

    # request = Request(url, headers=headers)

    # try:

    #     with urlopen(request, timeout=45) as response:

    #         status = response.status

    #         html = response.read().decode('utf-8', errors='replace')

    # except HTTPError as exc:

    #     status = exc.code

    #     html = exc.read().decode('utf-8', errors='replace')

  

    with requests.get(url, headers=headers, timeout=45) as response:

        status = response.status_code

        html = response.content.decode('utf-8', errors='replace')

    path = args.output / 'amazon.html'

    path.write_text(html, encoding='utf-8')

    print(f'HTTP 状态：{status}\nHTML 已保存到：{path.resolve()}')

    blocked = any(word in html.lower() for word in (

        'robot check', 'captcha', 'automated access', 'enter the characters you see below'))

    if blocked or status in (403, 429, 503):

        print('Amazon 返回了访问限制或验证码页；这是收到的真实页面，本次停止。')

    elif status >= 400:

        print('Amazon 返回错误页面，请查看保存的 HTML。')

    else:

        print('已收到页面。请打开 HTML，确认其中确实有商品搜索结果。')

  
  

if __name__ == '__main__':

    try:

        main()

    except Exception as exc:

        # 不打印请求头、请求体或密钥。

        print(f'运行失败：{type(exc).__name__}。请检查网络、接口权限和模型返回格式。')

        raise SystemExit(1)
```
